events: function(fetchInfo, successCallback, failureCallback) { fetch('./events.json') .then(response => response.json()) .then(data => { let eventos = data.map(item => { // LIMPIEZA SEGURA let servicio = (item.servicio || "") .toString() .toUpperCase() .replace(/\|/g, "") .trim(); let color = "#6c757d"; if (servicio.includes("REFRIGERIO")) color = "#0d6efd"; else if (servicio.includes("ALMUERZO")) color = "#198754"; else if (servicio.includes("CENA")) color = "#6f42c1"; else if (servicio.includes("PASABOCAS")) color = "#fd7e14"; else if (servicio.includes("DESAYUNO")) color = "#20c997"; else if (servicio.includes("TARDE JUEGOS")) color = "#ffc107"; let pack = parseInt(item.pack || 0); if (pack >= 100) color = "#dc3545"; return { title: item.cliente + " - " + item.servicio, start: item.date, backgroundColor: color, borderColor: color, extendedProps: item }; }); successCallback(eventos); }) .catch(error => { console.log("ERROR JSON:", error); failureCallback(error); }); }
